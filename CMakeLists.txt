cmake_minimum_required(VERSION 3.22)
project(zekals_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ANDROID)
  add_library(zekals_core SHARED native/core.cpp app/src/main/cpp/jni.cpp)
  target_link_options(zekals_core PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_library(zekals_core STATIC native/core.cpp)
  enable_testing()
  add_executable(core_test native/tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE zekals_core)
  # Assertions are part of the test in all build types.
  target_compile_options(core_test PRIVATE -UNDEBUG)
  add_test(NAME native_contract COMMAND core_test)
endif()
target_include_directories(zekals_core PUBLIC native/include)
target_compile_options(zekals_core PRIVATE -Wall -Wextra -Werror)
option(ZEKALS_SANITIZERS "Enable host address and undefined behavior sanitizers" OFF)
if(ZEKALS_SANITIZERS AND NOT ANDROID)
  target_compile_options(zekals_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(zekals_core PUBLIC -fsanitize=address,undefined)
endif()
